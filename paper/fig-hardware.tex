% The hardware figure for \S7. Kept in its own file so it can be redrawn without
% touching main.tex.
% Encoding: shading marks state that a commit installs atomically, vermillion is the
% control plane, blue is the enqueue walk and green the dequeue walk.
% The PE panels are stacked to match tree depth, so the commit's taps read top-down.
% The survivor link sits outside the PIFOs, per Zhiyuan: the PIFO implementation itself
% is unchanged, and the link only redirects a read. It is a block in the path's order
% rather than an arrow, because he asked that no path fork or double back.
% Each walk is one route shaped like an "n". Every block is tall enough to leave a clear
% band across its bottom, and the routes are drawn last so they run *in front of* the
% blocks, through that band, rather than showing only in the gaps between them.
% Enqueue rises at classify's centre and drops a third of the way into the PIFOs;
% dequeue rises two thirds of the way into them and drops at next hop's centre.
% Straight arrows drawn under the row were tried first and read as unattached to it.
\definecolor{ctlcol}{HTML}{D55E00}
\definecolor{enqcol}{HTML}{0072B2}
\definecolor{deqcol}{HTML}{009E73}
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  x=1cm, y=1cm,
  >={Stealth[length=4pt]},
  pebox/.style={draw=black!45, rounded corners=2.5pt, line width=0.6pt},
  unit/.style={draw=black!45, rounded corners=1.5pt, line width=0.5pt},
  txn/.style={unit, fill=black!8},
  bar/.style={rounded corners=0.6pt, line width=0.5pt},
  ul/.style={font=\scriptsize, inner sep=1pt, anchor=north, align=center},
  plab/.style={font=\scriptsize\itshape, text=black!55, inner sep=1pt},
  tl/.style={font=\scriptsize, inner sep=1pt},
  ctl/.style={draw=ctlcol, line width=0.75pt},
  enq/.style={draw=enqcol, line width=0.75pt},
  deq/.style={draw=deqcol, line width=0.75pt},
]

% ===================== transaction controller =====================
\draw[pebox] (0,4.30) rectangle (8.2,5.25);
\node[font=\scriptsize\bfseries, anchor=west, inner sep=1pt] at (0.18,4.70)
  {transaction controller};
% the instruction queue: the leading four cells are one transaction
\foreach \i in {0,...,8}{%
  \pgfmathsetmacro{\cx}{3.50+0.40*\i}%
  \ifnum\i<4
    \draw[unit, fill=ctlcol!18] (\cx,4.56) rectangle (\cx+0.40,4.84);
  \else
    \draw[unit] (\cx,4.56) rectangle (\cx+0.40,4.84);
  \fi}
\draw[decorate, decoration={brace, amplitude=3pt, raise=1pt},
      draw=black!55, line width=0.5pt] (3.50,4.86) -- (5.10,4.86);
\node[tl, anchor=south] at (4.30,5.00) {one transaction};
\draw[->, draw=black!55, line width=0.5pt] (7.70,4.70) -- (7.16,4.70);

% ===================== the commit, delay-aligned =====================
% One trunk out of the queue's head, tapping each level at its own instant.
\draw[ctl] (3.70,4.56) -- (3.70,4.10) -- (0.62,4.10) -- (0.62,0.39);
\node[tl, text=ctlcol, rotate=90, anchor=center] at (0.32,2.02) {delay-aligned commit};
\draw[ctl,->] (0.62,3.30) -- (1.24,3.30);
\draw[ctl,->] (0.62,1.23) -- (1.05,1.23);
\draw[ctl,->] (0.62,0.39) -- (1.05,0.39);
\node[tl, text=ctlcol, anchor=south] at (0.90,3.37) {\(t_0\)};
\node[tl, text=ctlcol, anchor=south] at (0.84,1.30) {\(t_1\)};
\node[tl, text=ctlcol, anchor=south] at (0.84,0.46) {\(t_2\)};

% ===================== the processing elements =====================
\draw[pebox] (1.05,1.82) rectangle (8.2,3.75);
\draw[pebox] (1.05,1.00) rectangle (8.2,1.46);
\draw[pebox] (1.05,0.16) rectangle (8.2,0.62);
\node[plab, anchor=south west] at (1.05,3.78) {PE, level 0};
\node[plab, anchor=south west] at (1.05,1.49) {PE, level 1};
\node[plab, anchor=south west] at (1.05,0.65) {PE, level 2};

% The deeper levels repeat level 0's blocks, with their contents left out.
% Same styles as level 0: the gray is a legend color, so it cannot vary by level.
\foreach \yb/\yt in {1.09/1.37, 0.25/0.53}{%
  \draw[txn]  (1.18,\yb) rectangle (2.53,\yt);
  \draw[unit] (2.73,\yb) rectangle (3.83,\yt);
  \draw[unit] (4.03,\yb) rectangle (5.28,\yt);
  \draw[unit] (5.48,\yb) rectangle (6.48,\yt);
  \draw[txn]  (6.68,\yb) rectangle (8.08,\yt);}

% ===================== level 0, drawn out =====================
\draw[txn]  (1.18,2.32) rectangle (2.53,3.66);
\draw[unit] (2.73,2.32) rectangle (3.83,3.66);
\draw[unit] (4.03,2.32) rectangle (5.28,3.66);

\node[ul] at (7.38,3.64) {next hop};

% double-buffered banks: solid is active, dashed is the copy the commit swaps in.
% Both mappers take the transaction commit, which Zhiyuan confirmed.
\draw[bar, draw=black!55, fill=black!16] (1.40,3.14) rectangle (2.45,3.34);

\draw[bar, draw=black!40, dashed]        (6.80,2.84) rectangle (7.96,3.04);

% rank-assignment logic is not double-buffered: its writes apply as they arrive
\draw[draw=black!50, fill=black!3, line width=0.5pt]
  (3.06,3.12) -- (3.46,3.03) -- (3.46,2.95) -- (3.06,2.86) -- cycle;

% two logical PIFOs in one PE, which is what spares the survivor a level of its own
\draw[bar, draw=black!55] (4.10,2.96) rectangle (4.60,3.22);
\draw[bar, draw=black!55] (4.72,2.96) rectangle (5.22,3.22);

% Enqueue: up at classify's centre, right along the band, down a third into the PIFOs.
